import os
import subprocess
import threading
import xml.etree.ElementTree as ET
import zipfile
import io
from flask import Flask, request, jsonify, send_from_directory, send_file

app = Flask(__name__, static_folder='public')

# Global state to track test progress
execution_state = {
    "status": "idle",
    "total": 0,
    "passed": 0,
    "failed": 0,
    "skipped": 0,
    "test_type": "None"
}


def run_robot_tests(test_type, region, env):
    """Executes the Robot Framework command and parses results."""
    global execution_state
    execution_state = {
        "status": "running",
        "total": 0, "passed": 0, "failed": 0, "skipped": 0,
        "test_type": test_type
    }

    base_path = os.path.abspath(os.path.dirname(__file__))
    results_dir = os.path.join(base_path, "results")
    tests_dir = os.path.join(base_path, "tests")

    # Ensure results directory exists
    os.makedirs(results_dir, exist_ok=True)

    # Robot Command: Includes variables passed from the UI
    cmd = [
        "python", "-m", "robot",
        "--outputdir", results_dir,
        "--variable", f"REGION:{region}",
        "--variable", f"ENV:{env}",
        "--include", test_type.lower(),
        tests_dir
    ]

    try:
        # Run the process
        subprocess.run(cmd, capture_output=True, text=True)

        # Parse output.xml to get live stats
        output_xml = os.path.join(results_dir, "output.xml")
        if os.path.exists(output_xml):
            tree = ET.parse(output_xml)
            root = tree.getroot()
            for stat in root.findall(".//statistics/total/stat"):
                if stat.text == 'All Tests':
                    execution_state.update({
                        "passed": int(stat.get('pass', 0)),
                        "failed": int(stat.get('fail', 0)),
                        "skipped": int(stat.get('skip', 0)),
                        "total": int(stat.get('pass', 0)) + int(stat.get('fail', 0)) + int(stat.get('skip', 0))
                    })
                    break
    except Exception as e:
        print(f"Execution Error: {e}")

    execution_state["status"] = "completed"


# --- ROUTES ---

@app.route('/')
def index():
    return send_from_directory('public', 'index.html')


@app.route('/chat', methods=['POST'])
def handle_chat():
    """Simple NLU to extract test parameters from user message."""
    user_text = request.json.get("message", "").lower()

    # Logic to identify intent
    test_type = next((t for t in ["sanity", "regression", "smoke"] if t in user_text), "sanity")
    region = next((r for r in ["us", "eu", "apac"] if r in user_text), "us")
    env = next((e for e in ["dev", "prod", "int", "stage"] if e in user_text), "dev")

    return jsonify({
        "type": "confirmation",
        "message": f"I've prepared a **{test_type.upper()}** run for **{region.upper()}** on **{env.upper()}**. Proceed?",
        "params": {
            "test_type": test_type.capitalize(),
            "region": region.upper(),
            "env": env.upper()
        }
    })


@app.route('/confirm_run', methods=['POST'])
def confirm_run():
    """Starts the test execution in a background thread."""
    params = request.json.get("params")
    thread = threading.Thread(
        target=run_robot_tests,
        args=(params['test_type'], params['region'], params['env'])
    )
    thread.start()
    return jsonify({"status": "started"})


@app.route('/status')
def get_status():
    """Polled by UI to update progress bars and stats."""
    return jsonify(execution_state)


@app.route('/results/<path:filename>')
def serve_results(filename):
    """Serves report.html and log.html from the results folder."""
    return send_from_directory('results', filename)


@app.route('/download_results')
def download_results():
    """Zips the entire results folder for the user."""
    base_path = os.path.abspath(os.path.dirname(__file__))
    results_dir = os.path.join(base_path, "results")

    memory_file = io.BytesIO()
    with zipfile.ZipFile(memory_file, 'w', zipfile.ZIP_DEFLATED) as zf:
        for root, _, files in os.walk(results_dir):
            for file in files:
                zf.write(os.path.join(root, file), file)

    memory_file.seek(0)
    return send_file(
        memory_file,
        mimetype='application/zip',
        as_attachment=True,
        download_name=f"Execution_Report_{execution_state['test_type']}.zip"
    )


@app.route('/raise_bug', methods=['POST'])
def raise_bug():
    """Simulates Jira Defect Creation."""
    data = request.json
    # In a real app, you would use 'requests' to POST to Jira's API here
    bug_id = "NEX-" + str(os.urandom(2).hex().upper())

    return jsonify({
        "status": "success",
        "bug_id": bug_id,
        "summary": data.get('title'),
        "attachments": ["report.html", "log.html", "screenshot.png"]
    })


if __name__ == '__main__':
    # Ensure the results folder exists on startup
    if not os.path.exists('results'):
        os.makedirs('results')
    app.run(port=8000, debug=True)

<script>
    let historyChart;
    let pendingParams = null;

    function toggleTheme() {
        const current = document.body.getAttribute('data-theme');
        document.body.setAttribute('data-theme', current === 'light' ? 'dark' : 'light');
    }

    function addLog(msg, isError = false) {
        const container = document.getElementById('log-container');
        const p = document.createElement('p');
        p.className = isError ? "text-rose-400" : "text-blue-300";
        p.innerText = `[${new Date().toLocaleTimeString()}] > ${msg}`;
        container.prepend(p);
    }

    function appendChat(sender, text) {
        const win = document.getElementById('chat-window');
        const div = document.createElement('div');
        div.className = sender === "User" ? "text-right" : "text-left";
        div.innerHTML = `<span class="opacity-40 text-[9px] uppercase font-bold">${sender}</span>
                         <div class="mt-1 p-2 rounded-lg ${sender==='User'?'bg-slate-700':'bg-blue-600/20 border border-blue-500/20'}">${text}</div>`;
        win.appendChild(div);
        win.scrollTop = win.scrollHeight;
    }

    async function sendChat() {
        const input = document.getElementById('chat-input');
        const msg = input.value; if(!msg) return;
        appendChat("User", msg);
        input.value = "";

        try {
            const res = await fetch('/chat', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({ message: msg })
            });
            const data = await res.json();
            if(data.type === 'confirmation') {
                pendingParams = data.params;
                document.getElementById('confirm-msg').innerHTML = data.message;
                document.getElementById('confirm-box').classList.remove('hidden');
            }
        } catch (e) { addLog("Chat connection failed.", true); }
    }

    async function finalConfirm(choice) {
        document.getElementById('confirm-box').classList.add('hidden');
        if(!choice) { appendChat("AI Agent", "Request cancelled."); return; }

        appendChat("AI Agent", "Processing... Robot engine active.");
        await fetch('/confirm_run', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({ params: pendingParams })
        });

        document.getElementById('stat-total').innerText = "0";
        document.getElementById('report-frame').src = "about:blank";
        document.getElementById('bug-box').classList.add('hidden');
        document.getElementById('download-btn').classList.add('hidden');

        beginPolling(pendingParams.test_type);
    }

    function beginPolling(type) {
        document.getElementById('active-suite-name').innerText = type.toUpperCase() + " RUN";
        const badge = document.getElementById('run-badge');
        badge.innerText = "Running";
        badge.className = "bg-emerald-600 text-[9px] px-2 py-1 rounded-md font-bold text-white animate-pulse uppercase";

        let prog = 0;
        const timer = setInterval(async () => {
            try {
                const res = await fetch('/status');
                const data = await res.json();
                if (data.status === 'running') {
                    if (prog < 95) prog += 5;
                    document.getElementById('main-p-bar').style.width = prog + "%";
                    document.getElementById('percentage-text').innerText = prog + "%";
                    document.getElementById('status-text').innerText = "Running Tests...";
                } else if (data.status === 'completed') {
                    clearInterval(timer);
                    document.getElementById('main-p-bar').style.width = "100%";
                    document.getElementById('percentage-text').innerText = "100%";
                    finalizeUI(data);
                }
            } catch(e) { clearInterval(timer); }
        }, 1500);
    }

    function finalizeUI(data) {
        document.getElementById('stat-total').innerText = data.total;
        document.getElementById('stat-passed').innerText = data.passed;
        document.getElementById('stat-failed').innerText = data.failed;
        document.getElementById('stat-skipped').innerText = data.skipped;

        const badge = document.getElementById('run-badge');
        badge.innerText = "Complete";
        badge.className = "bg-blue-600 text-[9px] px-2 py-1 rounded-md font-bold text-white uppercase";
        document.getElementById('status-text').innerText = "Ready";

        // Show download button
        document.getElementById('download-btn').classList.remove('hidden');
        document.getElementById('report-frame').src = "/results/report.html";

        historyChart.data.datasets[0].data = [data.passed, data.failed, data.skipped];
        historyChart.update();

        addLog(`Execution Finished: ${data.passed}P / ${data.failed}F.`);
        appendChat("AI Agent", `Execution finished. Results: ${data.passed} Passed, ${data.failed} Failed.`);

        if(data.failed > 0) document.getElementById('bug-box').classList.remove('hidden');
    }

    function openBugModal() {
        document.getElementById('bug-modal').classList.remove('hidden');
        const suite = document.getElementById('active-suite-name').innerText;
        document.getElementById('bug-title').value = `Bug: ${suite} failed in ${pendingParams.env}`;
        document.getElementById('bug-desc').value = `Summary: Detected ${document.getElementById('stat-failed').innerText} failures.\n\nContext:\nRegion: ${pendingParams.region}\nEnv: ${pendingParams.env}\nTime: ${new Date().toLocaleString()}`;
    }

    function closeBugModal() { document.getElementById('bug-modal').classList.add('hidden'); }

    async function submitBugToJira() {
        const payload = {
            title: document.getElementById('bug-title').value,
            parent_id: document.getElementById('bug-epic').value,
            priority: document.getElementById('bug-priority').value,
            description: document.getElementById('bug-desc').value
        };

        if(!payload.title || !payload.parent_id) {
            alert("Title and Parent ID are required.");
            return;
        }

        try {
            const res = await fetch('/raise_bug', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify(payload)
            });
            const data = await res.json();

            closeBugModal();
            document.getElementById('bug-box').classList.add('hidden');
            appendChat("AI Agent", `✅ **Bug ${data.bug_id} Created!** Linked to ${payload.parent_id}. Files attached.`);
            addLog(`Bug ${data.bug_id} logged to Jira.`);
        } catch (e) { addLog("Jira Error: Failed to push data.", true); }
    }

    function downloadResults() {
        addLog("Preparing results ZIP package...");
        window.location.href = '/download_results';
    }

    document.getElementById('chat-input').addEventListener('keypress', e => { if(e.key === 'Enter') sendChat(); });

    window.onload = () => {
        const ctx = document.getElementById('historyChart').getContext('2d');
        historyChart = new Chart(ctx, {
            type: 'doughnut',
            data: { labels: ['Pass', 'Fail', 'Skip'], datasets: [{ data: [0, 0, 0], backgroundColor: ['#10b981', '#f43f5e', '#f59e0b'], borderWidth: 0 }] },
            options: { cutout: '80%', plugins: { legend: { display: false } }, maintainAspectRatio: false }
        });
    };
