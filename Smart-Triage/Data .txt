<script src="https://cdn.jsdelivr.net/npm/sockjs-client@1/dist/sockjs.min.js"></script>
<script src="https://cdn.jsdelivr.net/npm/stompjs/lib/stomp.min.js"></script>

<script>
let historyChart;
let stompClient = null;

// 🔌 CONNECT WEBSOCKET
function connectSocket() {
    const socket = new SockJS("http://localhost:8080/ws");
    stompClient = Stomp.over(socket);

    stompClient.connect({}, function () {
        addLog("Connected to backend");

        // 🔥 LOG STREAM
        stompClient.subscribe("/topic/logs", function (msg) {
            addLog(msg.body);
        });

        // 🔥 STATUS STREAM
        stompClient.subscribe("/topic/status", function (msg) {
            const data = JSON.parse(msg.body);
            updateUI(data);
        });
    });
}

// 🧠 UPDATE UI FROM BACKEND
function updateUI(data) {
    document.getElementById('stat-total').innerText = data.total;
    document.getElementById('stat-passed').innerText = data.passed;
    document.getElementById('stat-failed').innerText = data.failed;
    document.getElementById('stat-skipped').innerText = data.skipped;

    // Progress calculation
    let percent = data.total === 0 ? 0 : Math.floor((data.passed + data.failed) / data.total * 100);
    document.getElementById('main-p-bar').style.width = percent + "%";
    document.getElementById('percentage-text').innerText = percent + "%";

    if (data.status === "running") {
        document.getElementById('status-text').innerText = "Executing...";
    }

    if (data.status === "completed") {
        finalizeUI(data);
    }
}

// 🚀 RUN TEST (CONNECTED TO SPRING BOOT)
async function runSuite() {

    // Reset UI
    document.getElementById('log-container').innerHTML = "";
    updateUI({ total: 0, passed: 0, failed: 0, skipped: 0 });

    const region = document.getElementById('sel-region').value;
    const env = document.getElementById('sel-env').value;

    const type = "sanity"; // or dynamic later

    document.getElementById('active-suite-name').innerText = type.toUpperCase();
    
    const badge = document.getElementById('run-badge');
    badge.innerText = "RUNNING";
    badge.className = "bg-emerald-600 text-[9px] px-2 py-1 rounded-md font-bold text-white animate-pulse";

    addLog(`Starting execution: ${region} ${env}`);

    // 🔥 CALL BACKEND
    await fetch("http://localhost:8080/run", {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify({
            region: region,
            env: env,
            type: type
        })
    });
}

// ✅ FINAL UI STATE
function finalizeUI(data) {
    const badge = document.getElementById('run-badge');
    badge.innerText = "FINISHED";
    badge.className = "bg-blue-600 text-[9px] px-2 py-1 rounded-md font-bold text-white";

    document.getElementById('status-text').innerText = "Completed";

    // Load Robot report
    document.getElementById('report-frame').src = "http://localhost:8080/results/report.html";

    historyChart.data.datasets[0].data = [data.passed, data.failed, data.skipped];
    historyChart.update();

    addLog(`Execution Completed ✅`);
}

// 🧾 LOG UI
function addLog(msg, isError = false) {
    const container = document.getElementById('log-container');

    const p = document.createElement('p');
    p.className = isError ? "text-rose-400" : "text-blue-300";
    p.innerText = `[${new Date().toLocaleTimeString()}] > ${msg}`;

    container.prepend(p);
}

// 🌗 THEME
function toggleTheme() {
    const current = document.body.getAttribute('data-theme');
    document.body.setAttribute('data-theme', current === 'light' ? 'dark' : 'light');
}

// 📊 CHART INIT
window.onload = () => {
    connectSocket(); // 🔥 IMPORTANT

    const ctx = document.getElementById('historyChart').getContext('2d');
    historyChart = new Chart(ctx, {
        type: 'doughnut',
        data: {
            labels: ['Pass', 'Fail', 'Skip'],
            datasets: [{
                data: [0, 0, 0],
                backgroundColor: ['#10b981', '#f43f5e', '#f59e0b'],
                borderWidth: 0
            }]
        },
        options: { 
            cutout: '80%', 
            plugins: { legend: { display: false } },
            maintainAspectRatio: false
        }
    });
};
</script>
