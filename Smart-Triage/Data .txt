<script>
    let historyChart;

    function toggleTheme() {
        const current = document.body.getAttribute('data-theme');
        document.body.setAttribute('data-theme', current === 'light' ? 'dark' : 'light');
    }

    function addLog(msg, isError = false) {
        const container = document.getElementById('log-container');
        const p = document.createElement('p');
        p.className = isError ? "text-rose-400" : "text-blue-300";
        p.innerText = `[${new Date().toLocaleTimeString()}] > ${msg}`;
        container.prepend(p);
    }

    // --- RESET AND EXECUTE ---
    async function runSuite() {
        // 1. Instant Zero Reset
        document.getElementById('stat-total').innerText = "0";
        document.getElementById('stat-passed').innerText = "0";
        document.getElementById('stat-failed').innerText = "0";
        document.getElementById('stat-skipped').innerText = "0";
        document.getElementById('main-p-bar').style.width = "0%";
        document.getElementById('percentage-text').innerText = "0%";
        
        historyChart.data.datasets[0].data = [0, 0, 0];
        historyChart.update();

        const region = document.getElementById('sel-region').value;
        const env = document.getElementById('sel-env').value;
        addLog(`Connecting to ${region} ${env} environment...`);

        try {
            const res = await fetch('/chat', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({ message: `Run Sanity for ${region} ${env}` })
            });
            const data = await res.json();
            if(data.type === 'success') beginPolling(data.test_type);
        } catch (e) {
            addLog("Error: Backend communication failed.", true);
        }
    }

    function beginPolling(type) {
        document.getElementById('active-suite-name').innerText = type.toUpperCase() + " RUN";
        const badge = document.getElementById('run-badge');
        badge.innerText = "RUNNING";
        badge.className = "bg-emerald-600 text-[9px] px-2 py-1 rounded-md font-bold text-white animate-pulse";

        let prog = 0;
        const bar = document.getElementById('main-p-bar');

        const timer = setInterval(async () => {
            const res = await fetch('/status');
            const data = await res.json();
            
            if (data.status === 'running') {
                if (prog < 90) prog += 7;
                bar.style.width = prog + "%";
                document.getElementById('percentage-text').innerText = prog + "%";
                document.getElementById('status-text').innerText = "Executing Robot Script...";
            } else if (data.status === 'completed') {
                clearInterval(timer);
                bar.style.width = "100%";
                document.getElementById('percentage-text').innerText = "100%";
                finalizeUI(data);
            }
        }, 1200);
    }

    function finalizeUI(data) {
        document.getElementById('stat-total').innerText = data.total;
        document.getElementById('stat-passed').innerText = data.passed;
        document.getElementById('stat-failed').innerText = data.failed;
        document.getElementById('stat-skipped').innerText = data.skipped;
        
        const badge = document.getElementById('run-badge');
        badge.innerText = "FINISHED";
        badge.className = "bg-blue-600 text-[9px] px-2 py-1 rounded-md font-bold text-white";
        document.getElementById('status-text').innerText = "Task Complete";
        document.getElementById('report-frame').src = "/results/report.html";

        historyChart.data.datasets[0].data = [data.passed, data.failed, data.skipped];
        historyChart.update();
        addLog(`Run Finished. Result: ${data.passed}P / ${data.failed}F`);
    }

    window.onload = () => {
        const ctx = document.getElementById('historyChart').getContext('2d');
        historyChart = new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: ['Pass', 'Fail', 'Skip'],
                datasets: [{
                    data: [0, 0, 0],
                    backgroundColor: ['#10b981', '#f43f5e', '#f59e0b'],
                    borderWidth: 0
                }]
            },
            options: { 
                cutout: '80%', 
                plugins: { legend: { display: false } },
                maintainAspectRatio: false
            }
        });
    };
</script>
