⚙️ 2. WebSocketConfig.java

package config;

import org.springframework.context.annotation.Configuration;
import org.springframework.messaging.simp.config.MessageBrokerRegistry;
import org.springframework.web.socket.config.annotation.*;

@Configuration
@EnableWebSocketMessageBroker
public class WebSocketConfig implements WebSocketMessageBrokerConfigurer {

    @Override
    public void configureMessageBroker(MessageBrokerRegistry config) {
        config.enableSimpleBroker("/topic"); // UI subscribes here
        config.setApplicationDestinationPrefixes("/app");
    }

    @Override
    public void registerStompEndpoints(StompEndpointRegistry registry) {
        registry.addEndpoint("/ws")
                .setAllowedOrigins("*")
                .withSockJS();
    }
}

🤖 4. RobotService.java (CORE LOGIC)

package service;

import lombok.RequiredArgsConstructor;
import model.ExecutionStatus;
import org.springframework.messaging.simp.SimpMessagingTemplate;
import org.springframework.stereotype.Service;

import java.io.BufferedReader;
import java.io.InputStreamReader;

@Service
@RequiredArgsConstructor
public class RobotService {

    private final SimpMessagingTemplate template;

    public void execute(String region, String env, String type) {

        try {
            template.convertAndSend("/topic/logs",
                    "🚀 Starting execution for " + region + " " + env);

            ProcessBuilder pb = new ProcessBuilder(
                    "robot",
                    "-d", "results",
                    "-v", "REGION:" + region,
                    "-v", "ENV:" + env,
                    "tests/" + type + ".robot"
            );

            Process process = pb.start();

            BufferedReader reader = new BufferedReader(
                    new InputStreamReader(process.getInputStream())
            );

            String line;
            int passed = 0, failed = 0;

            while ((line = reader.readLine()) != null) {

                // 🔥 Send log to UI
                template.convertAndSend("/topic/logs", line);

                if (line.contains("PASS")) passed++;
                if (line.contains("FAIL")) failed++;

                // 🔥 Send live status
                template.convertAndSend("/topic/status",
                        new ExecutionStatus(
                                "running",
                                passed + failed,
                                passed,
                                failed,
                                0
                        )
                );
            }

            process.waitFor();

            // ✅ Final update
            template.convertAndSend("/topic/status",
                    new ExecutionStatus(
                            "completed",
                            passed + failed,
                            passed,
                            failed,
                            0
                    )
            );

            template.convertAndSend("/topic/logs", "✅ Execution Completed");

        } catch (Exception e) {
            template.convertAndSend("/topic/logs",
                    "❌ Error: " + e.getMessage());
        }
    }
}

🎯 5. ExecutionController.java

package controller;

import lombok.RequiredArgsConstructor;
import org.springframework.web.bind.annotation.*;
import service.RobotService;

import java.util.Map;

@RestController
@RequiredArgsConstructor
@CrossOrigin("*")
public class ExecutionController {

    private final RobotService robotService;

    @PostMapping("/run")
    public String run(@RequestBody Map<String, String> req) {

        new Thread(() ->
                robotService.execute(
                        req.get("region"),
                        req.get("env"),
                        req.get("type")
                )
        ).start();

        return "Execution Started";
    }
}

🧠 6. ChatController.java (OPTIONAL AI ENTRY)

package controller;

import org.springframework.web.bind.annotation.*;

import java.util.Map;

@RestController
@CrossOrigin("*")
public class ChatController {

    @PostMapping("/chat")
    public Map<String, String> chat(@RequestBody Map<String, String> req) {

        String msg = req.get("message").toLowerCase();

        String type = "sanity";

        if (msg.contains("regression")) {
            type = "regression";
        }

        return Map.of(
                "type", "success",
                "test_type", type
        );
    }
}
